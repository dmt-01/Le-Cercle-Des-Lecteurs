@startuml Sequence Wishlist

actor Membre as U
participant "Frontend\nReact" as F
participant "Backend\nExpress" as B
participant "Base de données\nPostgreSQL" as DB

U -> F : Clique "Ajouter à la wishlist"
F -> F : Vérifie token JWT en localStorage
F -> B : POST /api/wishlist\n{ bookId }\nAuthorization: Bearer token

B -> B : Middleware JWT\nVérifie et décode le token

alt Token invalide ou expiré
  B --> F : 401 Unauthorized
  F --> U : Redirige vers connexion
else Token valide
  B -> DB : SELECT * FROM wishlist\nWHERE userId = ? AND bookId = ?
  DB --> B : Résultat

  alt Livre déjà dans la wishlist
    B --> F : 409 Conflict\n{ message: "Déjà dans la wishlist" }
    F --> U : Affiche notification
  else Livre absent
    B -> DB : INSERT INTO wishlist\n(userId, bookId, status, createdAt)
    DB --> B : Entrée créée
    B --> F : 201 Created\n{ message: "Ajouté à la wishlist" }
    F --> U : Bouton passe en "Dans ma wishlist"
  end
end

@enduml
