@startuml Sequence Connexion

actor Utilisateur as U
participant "Frontend\nReact" as F
participant "Backend\nExpress" as B
participant "Base de données\nPostgreSQL" as DB

U -> F : Saisit email + mot de passe
F -> F : Validation du formulaire
F -> B : POST /api/auth/login\n{ email, password }
B -> DB : SELECT * FROM users\nWHERE email = ?
DB --> B : Retourne l'utilisateur
B -> B : Bcrypt.compare(password, hash)

alt Mot de passe correct
  B -> B : JWT.sign(userId, secret)
  B --> F : 200 OK\n{ token, user }
  F -> F : Stocke le token
  F --> U : Redirige vers l'accueil
else Mot de passe incorrect
  B --> F : 401 Unauthorized\n{ message: "Identifiants invalides" }
  F --> U : Affiche message d'erreur
end

@enduml
