:L'utilisateur consulte la fiche d'un livre;

if (Utilisateur connecté ?) then (non)
  :Redirection vers la page de connexion;
  stop
else (oui)
  if (A déjà critiqué ce livre ?) then (oui)
    :Affiche "Vous avez déjà noté ce livre";
    :Propose de modifier la critique existante;
  else (non)
  endif
  :Clique sur "Rédiger / Modifier une critique";
  :Saisit une note (1 à 5);
  :Saisit le texte de la critique;
  if (Contenu valide ?) then (non)
    :Affiche les erreurs;
    stop
  else (oui)
    :Envoi au backend;
    :Sauvegarde en BDD;
    :Recalcul de la note moyenne du livre;
    :Attribution badge si première critique;
    :Affichage de la critique publiée;
    stop
  endif
endif
